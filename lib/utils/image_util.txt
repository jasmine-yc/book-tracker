// 儲存圖片方法
Future<void> _pickImage() async{
  final picker =ImagePicker();
  final pickedFile = await picker.pickImage(source: ImageSource.gallery);

  if(pickedFile != null) {
    final savedPath = await saveCoverImage(File(pickedFile.path)); // 使用工具函式

    if(savedPath !=null){
      setState((){
        _coverImageFile = File(savedPath);// 用新的路徑重新顯示圖片
      });
    }else {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('圖片儲存失敗，請重試')),
      );
    } 
  }    
}
