
void _addTag(String tag) {
    final trimmedTag = tag.trim();
    if(trimmedTag.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('標籤不得為空')),
      );
      _tagController.clear();  // 清空輸入框
      return;
    } // 標籤不得為空
    if(trimmedTag.length > 10) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('標籤"$trimmedTag"過長（最多 10 字）')),
      );
      _tagController.clear();  // 清空輸入框
      return;
    } // 標籤不得過長
    bool tagExists(String newTag) {
    final normalizedNewTag = normalize(newTag);
    return _selectedTags.any((tag) => normalize(tag) == normalizedNewTag);
    }
    // 應對比大小寫
    if(tagExists(trimmedTag)) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('標籤"$trimmedTag"已存在（含大小寫）')),
      );
      _tagController.clear();  // 清空輸入框
      return;
    } // 標籤不得重複
    

    if(_selectedTags.length >= 5) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('標籤數量不得超過 5 個')),
      );
      _tagController.clear();  // 清空輸入框
      return;
    }
    setState((){
      _selectedTags.add(trimmedTag); // 加入已選標籤
      _tagController.clear(); //清空輸入框
    });
  }

  void _removeTag(String tag) {
    setState((){
      _selectedTags.remove(tag);
    });
  }
